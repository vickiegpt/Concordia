cmake_minimum_required(VERSION 3.22)
project(pacc_sequence_capacity LANGUAGES CXX)
set(LLAMA_MAX_SEQ 512 CACHE STRING "Independent sequence capacity")
foreach(required LLAMA_SOURCE GGML_SOURCE GGML_BIN PROBE_SOURCE HARNESS_SOURCE JSON_INCLUDE)
  if(NOT DEFINED ${required})
    message(FATAL_ERROR "Missing ${required}")
  endif()
endforeach()
file(GLOB model_sources CONFIGURE_DEPENDS "${LLAMA_SOURCE}/src/models/*.cpp")
file(GLOB core_sources CONFIGURE_DEPENDS "${LLAMA_SOURCE}/src/*.cpp")
add_library(llama SHARED ${core_sources} ${model_sources})
target_include_directories(llama PUBLIC "${LLAMA_SOURCE}/include" "${GGML_SOURCE}/include"
                           PRIVATE "${LLAMA_SOURCE}/src")
target_compile_features(llama PRIVATE cxx_std_17)
target_compile_definitions(llama PRIVATE LLAMA_BUILD LLAMA_MAX_SEQ=${LLAMA_MAX_SEQ}
  LLAMA_COMMIT="d222767-seq512" LLAMA_VERSION="0.3.0-dev"
  GGML_BACKEND_SHARED GGML_SHARED GGML_USE_CPU GGML_USE_CUDA)
target_compile_definitions(llama PUBLIC LLAMA_SHARED)
target_link_libraries(llama PUBLIC "${GGML_BIN}/libggml.so" "${GGML_BIN}/libggml-base.so"
  "${GGML_BIN}/libggml-cpu.so" "${GGML_BIN}/libggml-cuda.so")
set_target_properties(llama PROPERTIES VERSION 0.3.0 SOVERSION 0
  LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin" BUILD_RPATH "${GGML_BIN}")
add_executable(qwen-capacity-probe "${PROBE_SOURCE}")
add_executable(qwen-multiseq-v4 "${HARNESS_SOURCE}")
foreach(target qwen-capacity-probe qwen-multiseq-v4)
  target_compile_features(${target} PRIVATE cxx_std_17)
  target_include_directories(${target} PRIVATE "${JSON_INCLUDE}")
  target_link_libraries(${target} PRIVATE llama)
  target_link_options(${target} PRIVATE "-Wl,--allow-shlib-undefined")
  set_target_properties(${target} PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
endforeach()
